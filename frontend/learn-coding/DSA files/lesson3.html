def is_palindrome(s):
    left = 0
    right = len(s) - 1

    while left < right:
        if s[left] != s[right]:
            return False

        left += 1
        right -= 1

    return True


print(is_palindrome("racecar"))
Java 
static boolean isPalindrome(String s) {

    int left = 0;
    int right = s.length() - 1;

    while (left < right) {

        if (s.charAt(left) != s.charAt(right)) {
            return false;
        }

        left++;
        right--;
    }

    return true;
}

Anagrams

Two strings are anagrams when they contain the same character counts, assuming the same normalization rules.

LISTEN
⇄
SILENT

Frequency Map Method

Python 
def is_anagram(a, b):

    if len(a) != len(b):
        return False

    freq = {}

    for ch in a:
        freq[ch] = freq.get(ch, 0) + 1

    for ch in b:
        if ch not in freq:
            return False

        freq[ch] -= 1

        if freq[ch] < 0:
            return False

    return True
Java 
static boolean isAnagram(
    String a, String b) {

    if (a.length() != b.length())
        return false;

    HashMap<Character, Integer> map =
        new HashMap<>();

    for (char ch : a.toCharArray()) {
        map.put(
            ch,
            map.getOrDefault(ch, 0) + 1
        );
    }

    for (char ch : b.toCharArray()) {

        if (!map.containsKey(ch))
            return false;

        map.put(ch, map.get(ch) - 1);

        if (map.get(ch) < 0)
            return false;
    }

    return true;
}
Another approach is sorting both strings and comparing the sorted results. That usually costs O(n log n).

Longest substring without repeating characters

Python 
def longest_unique(s):

    seen = set()

    left = 0
    best = 0

    for right in range(len(s)):

        while s[right] in seen:
            seen.remove(s[left])
            left += 1

        seen.add(s[right])

        best = max(
            best,
            right - left + 1
        )

    return best

print(longest_unique("abcabcbb"))
Java 
static int longestUnique(String s) {

    HashSet<Character> set =
        new HashSet<>();

    int left = 0;
    int best = 0;

    for (int right = 0;
         right < s.length();
         right++) {

        while (set.contains(s.charAt(right))) {
            set.remove(s.charAt(left));
            left++;
        }

        set.add(s.charAt(right));

        best = Math.max(
            best,
            right - left + 1
        );
    }

    return best;
}

KMP Algorithm

KMP avoids repeatedly comparing characters that are already known to match.

A B A B C
0 0 1 2 0
LPS
LPS = Longest Proper Prefix which is also a Suffix.

Naive

Repeated comparisons can occur.

KMP

Previous matching information is reused.

Python — LPS 
def build_lps(pattern):

    lps = [0] * len(pattern)

    length = 0
    i = 1

    while i < len(pattern):

        if pattern[i] == pattern[length]:
            length += 1
            lps[i] = length
            i += 1

        elif length != 0:
            length = lps[length - 1]

        else:
            lps[i] = 0
            i += 1

    return lps
Java — LPS 
static int[] buildLPS(String p) {

    int[] lps = new int[p.length()];

    int length = 0;
    int i = 1;

    while (i < p.length()) {

        if (p.charAt(i) == p.charAt(length)) {

            length++;
            lps[i] = length;
            i++;

        } else if (length != 0) {

            length = lps[length - 1];

        } else {

            lps[i] = 0;
            i++;
        }
    }

    return lps;
}

    if (a.length() != b.length())

    return (a + a).contains(b);

String Compression

aaabbcccc
→
a3b2c4

Run-Length Encoding

Scan consecutive equal characters and store each character together with its run length.

Python 
def compress(s):

    result = []
    i = 0

    while i < len(s):

        j = i

        while j < len(s) and s[j] == s[i]:
            j += 1

        result.append(
            s[i] + str(j - i)
        )

        i = j

    return "".join(result)

print(compress("aaabbcccc"))
Java 
static String compress(String s) {

    StringBuilder result =
        new StringBuilder();

    int i = 0;

    while (i < s.length()) {

        int j = i;

        while (j < s.length()
               && s.charAt(j) == s.charAt(i)) {
            j++;
        }

        result.append(s.charAt(i));
        result.append(j - i);

        i = j;
    }

    return result.toString();
}

String Interview Questions

What is a substring?

A contiguous sequence of characters inside a string.

What is a subsequence?

A sequence formed by deleting zero or more characters without changing the relative order of the remaining characters.

String vs character array?

A character array stores individual character elements, while a string is a language-level abstraction for text with its own operations and representation rules.

Why are Java Strings immutable?

Java String objects cannot be changed after creation. This allows them to be safely shared and enables several implementation optimizations.

How do you check an anagram?

Compare character frequencies or sort both strings and compare the results.

How does sliding window work on strings?

Maintain a contiguous range and move the left and right boundaries while maintaining the required condition.

When should you use HashMap?

When fast average lookup, counting, mapping or frequency tracking is useful.

What is KMP?

A pattern matching algorithm that uses prefix information to avoid unnecessary comparisons.

What is LPS?

Longest Proper Prefix which is also a Suffix. It is the preprocessing information used by KMP.

What is a rolling hash?

A hash technique that updates the hash of a moving window efficiently instead of recomputing it from scratch.

KMP vs Rabin-Karp?

KMP uses prefix-function information, while Rabin-Karp uses rolling hashes to compare candidate windows.
