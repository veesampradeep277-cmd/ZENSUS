Project Example — Smart Calculator

Start with a simple project and gradually introduce professional structure.

calculator.c 
#include <stdio.h>

int add(int a, int b)
{
    return a + b;
}

int subtract(int a, int b)
{
    return a - b;
}

int multiply(int a, int b)
{
    return a * b;
}

int main()
{
    int choice;
    int a, b;

    printf("===== SMART CALCULATOR =====\n");

    printf("1. Addition\n");
    printf("2. Subtraction\n");
    printf("3. Multiplication\n");
    printf("4. Division\n");

    printf("Enter choice: ");
    scanf("%d", &choice);

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    switch(choice)
    {
        case 1:
            printf("Result = %d\n", add(a, b));
            break;

        case 2:
            printf("Result = %d\n", subtract(a, b));
            break;

        case 3:
            printf("Result = %d\n", multiply(a, b));
            break;

        case 4:

            if(b == 0)
            {
                printf("Error: Division by zero.\n");
            }
            else
            {
                printf("Result = %.2f\n",
                       (double)a / b);
            }

            break;

        default:
            printf("Invalid choice.\n");
    }

    return 0;
}

Complete Starter Version

student_records.c 
#include <stdio.h>
#include <stdlib.h>

struct Student
{
    int roll;
    char name[50];
    float marks;
};

void addStudent()
{
    FILE *fp;

    struct Student s;

    fp = fopen("students.dat", "ab");

    if(fp == NULL)
    {
        perror("Unable to open file");
        return;
    }

    printf("\nEnter Roll Number: ");
    scanf("%d", &s.roll);

    printf("Enter Name: ");
    scanf(" %49[^\n]", s.name);

    printf("Enter Marks: ");
    scanf("%f", &s.marks);

    if(fwrite(&s, sizeof(s), 1, fp) != 1)
    {
        printf("Error writing record.\n");
    }
    else
    {
        printf("Student added successfully.\n");
    }

    fclose(fp);
}

void displayStudents()
{
    FILE *fp;

    struct Student s;

    fp = fopen("students.dat", "rb");

    if(fp == NULL)
    {
        perror("Unable to open file");
        return;
    }

    printf("\n===== STUDENT RECORDS =====\n");

    while(fread(&s, sizeof(s), 1, fp) == 1)
    {
        printf("\nRoll  : %d", s.roll);
        printf("\nName  : %s", s.name);
        printf("\nMarks : %.2f\n", s.marks);
    }

    fclose(fp);
}

void searchStudent()
{
    FILE *fp;

    struct Student s;

    int roll;
    int found = 0;

    fp = fopen("students.dat", "rb");

    if(fp == NULL)
    {
        perror("Unable to open file");
        return;
    }

    printf("\nEnter roll number to search: ");
    scanf("%d", &roll);

    while(fread(&s, sizeof(s), 1, fp) == 1)
    {
        if(s.roll == roll)
        {
            printf("\nStudent Found!");

            printf("\nRoll  : %d", s.roll);
            printf("\nName  : %s", s.name);
            printf("\nMarks : %.2f\n", s.marks);

            found = 1;
            break;
        }
    }

    if(!found)
    {
        printf("\nStudent not found.\n");
    }

    fclose(fp);
}

int main()
{
    int choice;

    do
    {
        printf("\n\n==============================");
        printf("\n STUDENT RECORD MANAGEMENT");
        printf("\n==============================");

        printf("\n1. Add Student");
        printf("\n2. Display Students");
        printf("\n3. Search Student");
        printf("\n4. Exit");

        printf("\n\nEnter choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1:
                addStudent();
                break;

            case 2:
                displayStudents();
                break;

            case 3:
                searchStudent();
                break;

            case 4:
                printf("\nThank you!\n");
                break;

            default:
                printf("\nInvalid choice.\n");
        }

    }while(choice != 4);

    return 0;
}

Important binary-file note:
Writing a raw struct with fwrite() is convenient for a learning project, but it is not a universally portable file format. Structure padding, type representation and byte order can differ between systems. Avoid storing pointers inside such persistent structures.

Alternative — Text-Based Student Records

Text files are easier to inspect and exchange manually.

students.txt 
101,Pradeep,91.5
102,Rahul,87.0
103,Ananya,94.5
write_record.c 
#include <stdio.h>

int main()
{
    FILE *fp;

    fp = fopen("students.txt", "a");

    if(fp == NULL)
    {
        perror("fopen");
        return 1;
    }

    fprintf(fp,
            "%d,%s,%.2f\n",
            101,
            "Pradeep",
            91.5);

    fclose(fp);

    return 0;
}
Choosing the format: Text files are human-readable and convenient for simple interchange. Binary files can be compact and efficient for controlled applications, but raw structures are not a portable serialization format.

Common Project Mistakes

These mistakes make otherwise working programs fragile.

❌ Everything in main()

Large programs become difficult to read, test and maintain.

❌ No Input Validation

Unexpected input can cause incorrect results or unsafe behavior.

❌ Ignoring File Errors

Always check whether fopen() succeeded and check important read/write operations.

❌ Buffer Overflow

Never read unlimited text into a fixed-size character array.

❌ Magic Numbers

Prefer meaningful constants and named values where appropriate.

❌ No Testing

A project is not complete simply because it compiles.

Never assume compilation means correctness.
A C program can compile successfully and still contain undefined behavior, incorrect logic, memory errors or incorrect file handling.

Interview Questions

Questions commonly asked when discussing C projects.

Large programs become difficult to read, test and maintain. Functions allow responsibilities to be separated into smaller reusable units.
A structure groups related fields of potentially different types into one object. For example, a Student structure can contain roll number, name and marks.
Normal variables and arrays exist during program execution. Files allow information to persist after the program terminates.
Text storage represents information using textual characters, making it human-readable. Binary storage works with bytes and is often convenient for structured program data, but raw binary structures can have portability limitations.
Add update and delete operations, input validation, duplicate-roll detection, sorting, better searching, persistent storage, error handling and a clean modular architecture.
Check whether the returned pointer is NULL, report the failure appropriately, for example with perror(), and avoid using the invalid stream.
